# st
```
+--+--+--+--+--+--+ 
| R| R| R| B| B| B|  
+--+--+--+--+--+--+
        R  L
```

关键：循环不变量
- L - 1 始终是红色
- R + 1 始终是蓝色
- R + 1 = L

如果所有元素都小于要查找的元素，则L就为数组长度

理解二分，请牢记区间的定义！区间内的数（下标）都是还未确定与 target 的大小关系的，有的是 < target，有的是 ≥ target；区间外的数（下标）都是确定与 target 的大小关系的。

# 在排序数组中查找元素的第一个和最后一个位置

https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/solution/er-fen-cha-zhao-zong-shi-xie-bu-dui-yi-g-t9l9/

给你一个按照非递减顺序排列的整数数组 nums，和一个目标值 target。请你找出给定目标值在数组中的开始位置和结束位置。
如果数组中不存在目标值 target，返回 [-1, -1]。
你必须设计并实现时间复杂度为 O(log n) 的算法解决此问题。

示例 1：
输入：nums = [5,7,7,8,8,10], target = 8
输出：[3,4]
示例 2：
输入：nums = [5,7,7,8,8,10], target = 6
输出：[-1,-1]

## 闭区间写法
```cpp
class Solution {
    // lower_bound 返回最小的满足 nums[i] >= target 的下标 i
    // 如果数组为空，或者所有数都 < target，则返回 nums.size()
    // 要求 nums 是非递减的，即 nums[i] <= nums[i + 1]
    int lower_bound(vector<int>& nums, int target) {
        int left = 0, right = (int) nums.size() - 1; // 闭区间 [left, right]
        while (left <= right) { // 区间不为空
            // 循环不变量：
            // nums[left-1] < target
            // nums[right+1] >= target
            int mid = left + (right - left) / 2;
            if (nums[mid] >= target) {
                right = mid - 1; // 范围缩小到 [left, mid-1]
            } else {
                left = mid + 1; // 范围缩小到 [mid+1, right]
            }
        }
        // 循环结束后 left = right+1
        // 此时 nums[left-1] < target 而 nums[left] = nums[right+1] >= target
        // 所以 left 就是第一个 >= target 的元素下标
        return left;
    }

public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int start = lower_bound(nums, target);
        if (start == nums.size() || nums[start] != target) {
            return {-1, -1}; // nums 中没有 target
        }
        // 如果 start 存在，那么 end 必定存在
        int end = lower_bound(nums, target + 1) - 1;
        return {start, end};
    }
};
```

## 开区间写法

```cpp

    // 如果数组为空，或者所有数都 < target，则返回 nums.size()

    int lower_bound(vector<int>& nums, int target) {
        int left = -1, right = nums.size(); // 开区间 (left, right)

            // 循环不变量：
            // nums[left] < target
            // nums[right] >= target

            if (nums[mid] >= target) {
                right = mid; // 范围缩小到 (left, mid)
            } else {
                left = mid; // 范围缩小到 (mid, right)
            }
        }
        // 循环结束后 left+1 = right
        // 此时 nums[left] < target 而 nums[right] >= target
        // 所以 right 就是第一个 >= target 的元素下标
        return right;
    }

public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int start = lower_bound(nums, target);
        if (start == nums.size() || nums[start] != target) {
            return {-1, -1}; // nums 中没有 target
        }
        // 如果 start 存在，那么 end 必定存在
        int end = lower_bound(nums, target + 1) - 1;
        return {start, end};
    }
};
```

## 库函数写法

```cpp
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int start = ranges::lower_bound(nums, target) - nums.begin();
        if (start == nums.size() || nums[start] != target) {
            return {-1, -1};
        }
        int end = ranges::upper_bound(nums, target) - nums.begin() - 1;
        return {start, end};
    }
};
```

## 总结

|需求|写法|如果不存在|
|:---|:---|:---|
|≥x 的第一个元素的下标|``lowerBound(nums,x)``|结果为n|
|>x 的第一个元素的下标|``lowerBound(nums,x+1)``|结果为n|
|≤x 的最后一个元素的下标|``lowerBound(nums,x+1)-1``|结果为-1|
|<x 的最后一个元素的下标|``lowerBound(nums,x)-1``|结果为-1|

# 正整数和负整数的最大计数

https://leetcode.cn/problems/maximum-count-of-positive-integer-and-negative-integer/

```cpp
class Solution {

   int lower_bound(vector<int>& nums, int target){
    int left = -1,right = nums.size();
    while (left + 1 < right){
        int mid = left + (right - left)/2;
        if (nums[mid] >= target){
            right = mid;
        }
        else{
            left = mid;
        }
    }
    return right;
   }

public:
    int maximumCount(vector<int>& nums) {
        int Pcnt = 0;
        int Ncnt = 0;
        // 计算正整数的个数(>0)
        Pcnt = lower_bound(nums,1);
        if (Pcnt == nums.size()){ //没有正整数
            Pcnt = 0;
        } 
        else{
            Pcnt = nums.size() - Pcnt;
        }

        // 计算负整数的个数(<0)
        Ncnt = lower_bound(nums,0) - 1;
        if (Ncnt == -2){  // 没有负整数
            Ncnt = 0;
        }
        else{
            Ncnt = Ncnt + 1;
        }
        
        return max(Ncnt,Pcnt);
    }
};
```

```cpp
class Solution {
public:
    int lowerBound(vector<int>& nums, int val) {
        int l = 0, r = nums.size();
        while (l < r) {
            int m = (l + r) / 2;
            if (nums[m] >= val) {
                r = m;
            } else {
                l = m + 1;
            }
        }
        return l;
    }

    int maximumCount(vector<int>& nums) {
        int pos1 = lowerBound(nums, 0);
        int pos2 = lowerBound(nums, 1);
        return max(pos1, (int) nums.size() - pos2);
    }
};
```

# 咒语和药水的成功对数

https://leetcode.cn/problems/successful-pairs-of-spells-and-potions/

```cpp
class Solution {
public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {
        vector<int> temp;
        long long int a;
        ranges::sort(potions);
        for (int n = 0;n < spells.size(); n++){
            // 向上取整
            a = (success + spells[n] - 1) / spells[n];
            int start = ranges::lower_bound(potions,a) - potions.begin();
            start = potions.size() - start;
            temp.push_back(start);
        }
        return temp;
    }
};
```

# 两个数组间的距离值

https://leetcode.cn/problems/find-the-distance-value-between-two-arrays/

```cpp
class Solution {
public:
    int findTheDistanceValue(vector<int>& arr1, vector<int>& arr2, int d) {
        int cnt = 0;
        ranges::sort(arr2);
        for (int x : arr1){
            auto it = ranges::lower_bound(arr2, x - d);
            if (it == arr2.end() || *it > x + d){
                cnt++;
            }
        }
        return cnt;
    }
};
```

# 区间内查询数字的频率

https://leetcode.cn/problems/range-frequency-queries/

```cpp
class RangeFreqQuery {
    unordered_map<int, vector<int>> pos;

public:
    RangeFreqQuery(vector<int>& arr) {
        for (int i = 0; i < arr.size(); i++) {
            pos[arr[i]].push_back(i);
        }
    }

    int query(int left, int right, int value) {
        // 不推荐写 a = pos[value]，如果 value 不在 pos 中会插入 value
        auto it = pos.find(value);
        if (it == pos.end()) {
            return 0;
        }
        auto& a = it->second;
        return ranges::upper_bound(a, right) - ranges::lower_bound(a, left);
    }
};
```

# 统计公平数对的数目

https://leetcode.cn/problems/count-the-number-of-fair-pairs/

```CPP
class Solution {
public:
    long long countFairPairs(vector<int>& nums, int lower, int upper) {
        ranges::sort(nums);
        long long ans = 0;
        for (int j = 0; j < nums.size(); j++){
            // lower - nums[j] <= nums[i] <= upper - nums[j]
            auto r = lower_bound(nums.begin(), nums.begin() + j, upper - nums[j] + 1) - 1;
            auto l = lower_bound(nums.begin(), nums.begin() + j, lower - nums[j]);
            ans += r - l + 1;
        }
        return ans;
    }
};
```

# 爱吃香蕉的珂珂

https://leetcode.cn/problems/koko-eating-bananas/

class Solution {
public:
    int minEatingSpeed(vector<int>& piles, int h) {
        auto check = [&](int k) -> bool{
            int sum = piles.size();
            for (int p : piles){
                sum += (p - 1) / k;
                if (sum > h){
                    return false;
                }
            }
            return true;
        };

        int left = 0; // 循环不变量
        int right = ranges::max(piles); // 循环不变量
        while (left + 1 < right){ // 开区间不变量
            int mid = left + (right - left) / 2;
            (check(mid) ? right : left) = mid;
        }
        return right; // 最小的 true
    }
};
